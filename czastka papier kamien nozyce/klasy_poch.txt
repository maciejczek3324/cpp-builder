komar *kom;


komar::komar()
      {
      x=random(Sx);
      y=random(Sy);
      vx=random(7)-3;
      vy=random(9)-4;
      zywy=1;

      k=new TShape(Form1);
      k->Parent=Form1;
      k->Left=x;
      k->Top=y;
      k->Width=5;
      k->Height=5;
      k->Shape=stCircle;
      }


void komar::ruch()
      {
      x+=vx;
      y+=vy;
      if(x>Sx) vx=-vx;
      if(x<0) vx=-vx;
      if(y>Sy) vy=-vy;
      if(y<0) vy=-vy;
      k->Left=x;
      k->Top=y;
      }


void komar::smierc()
  {
  if(random(1000)<6) zywy=0;
  }
